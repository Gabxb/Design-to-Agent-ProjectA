client = OpenAI(
    api_key="sk-xxx",
    base_url="https://api.deepseek.com/v1"
)

def calculator(expression):
    """评估数学表达式"""
    try:
        return str(eval(expression))
    except:
        return "Error"

TOOLS = [{
    "type": "function",
    "function": {
        "name": "calculator",
        "description": "计算数学表达式",
        "parameters": {
            "type": "object",
            "properties": {
                "expression": {"type": "string"}
            },
            "required": ["expression"]
        }
    }
}]

def simple_agent(user_message, max_steps=5):
    messages = [{"role": "user", "content": user_message}]

    for _ in range(max_steps):
        response = client.chat.completions.create(
            model="deepseek-v4",
            messages=messages,
            tools=TOOLS
        )
        msg = response.choices[0].message
        messages.append(msg)

        if not msg.tool_calls:
            return msg.content  # 最终回答

        # 执行工具调用
        for tc in msg.tool_calls:
            if tc.function.name == "calculator":
                import json
                args = json.loads(tc.function.arguments)
                result = calculator(args["expression"])
                messages.append({
                    "role": "tool",
                    "tool_call_id": tc.id,
                    "content": result
                })
    return "Max steps reached"

3. 实现工具注册系统 ⭐⭐

# 要求：可动态注册/注销工具，支持参数校验
import inspect
from typing import Callable, get_type_hints

class ToolRegistry:
    def __init__(self):
        self._tools = {}

    def register(self, func: Callable):
        """注册一个函数为工具"""
        hints = get_type_hints(func)
        schema = {
            "type": "object",
            "properties": {},
            "required": []
        }
        for name, typ in hints.items():
            if name == "return":
                continue
            schema["properties"][name] = {
                "type": "string" if typ == str else "number"
            }
            schema["required"].append(name)

        self._tools[func.__name__] = {
            "function": func,
            "schema": {
                "name": func.__name__,
                "description": func.__doc__ or "",
                "parameters": schema
            }
        }

    def get_tool_definitions(self):
        """返回 OpenAI 格式的工具定义"""
        return [{"type": "function", "function": t["schema"]}
                for t in self._tools.values()]

    def execute(self, name, args):
        """执行指定工具"""
        if name not in self._tools:
            raise ValueError(f"Tool {name} not found")
        return self._tools[name]["function"](**args)

# 使用示例
registry = ToolRegistry()

@registry.register
def get_weather(city: str) -> str:
    """查询指定城市的天气"""
    return f"{city}: Sunny, 25°C"

@registry.register
def calculate(expression: str) -> str:
    """计算数学表达式"""
    return str(eval(expression))

        else:

            "role": "user",

        })

5. 实现带超时和重试的工具调用 ⭐⭐

import asyncio
import functools

def with_retry(max_retries=2, timeout=10):
    """工具调用装饰器：超时重试"""
    def decorator(func):
        @functools.wraps(func)
        async def wrapper(*args, **kwargs):
            for attempt in range(max_retries + 1):
                try:
                    return await asyncio.wait_for(
                        func(*args, **kwargs),
                        timeout=timeout
                    )
                except asyncio.TimeoutError:
                    if attempt < max_retries:
                        print(f"Retry {attempt + 1}/{max_retries}")
                    else:
                        raise
            return None
        return wrapper
    return decorator

@with_retry(max_retries=2, timeout=5)
async def search_knowledge_base(query: str):
    """模拟知识库搜索"""
    await asyncio.sleep(3)  # 模拟网络延迟
    return [f"Result for: {query}"]

from openai import OpenAI
import faiss
import numpy as np

client = OpenAI(
    api_key="sk-xxx",
    base_url="https://api.deepseek.com/v1"
)

class SimpleRAG:
    def __init__(self):
        self.documents = []
        self.index = None

    def get_embedding(self, text):
        response = client.embeddings.create(
            model="text-embedding-3-small",
            input=text
        )
        return np.array(response.data[0].embedding)

    def add_documents(self, docs):
        self.documents = docs
        embeddings = [self.get_embedding(d) for d in docs]

        dim = len(embeddings[0])
        self.index = faiss.IndexFlatL2(dim)
        self.index.add(np.array(embeddings))

    def search(self, query, k=3):
        query_vec = self.get_embedding(query).reshape(1, -1)
        distances, indices = self.index.search(query_vec, k)
        return [self.documents[i] for i in indices[0]]

7. 实现带重排序的检索 ⭐⭐

def search_with_rerank(query, retriever, reranker, k=20, top_n=3):
    # 1. 粗排：向量检索 Top-K
    candidates = retriever.get_relevant_documents(query, k=k)

    # 2. 精排：用 Reranker 打分
    scores = []
    for doc in candidates:
        score = reranker.compute_score(query, doc.page_content)
        scores.append((score, doc))

    # 3. 返回 Top-N
    scores.sort(key=lambda x: x[0], reverse=True)
    return [doc for _, doc in scores[:top_n]]

8. 实现文档 Chunk 分割 ⭐

def chunk_document(text, chunk_size=512, overlap=50):
    """固定长度切分，带重叠"""
    chunks = []
    start = 0

    while start < len(text):
        end = start + chunk_size
        chunk = text[start:end]

        # 尽量在句子边界断开
        if end < len(text):
            last_period = chunk.rfind('。')
            if last_period > chunk_size * 0.5:
                end = start + last_period + 1

        chunks.append(text[start:end])
        start = end - overlap

    return chunks

9. 实现 Hybrid Search（向量 + BM25） ⭐⭐

from rank_bm25 import BM25Okapi

def hybrid_search(query, vector_store, documents, k=5, alpha=0.6):
    # 向量检索
    vec_results = vector_store.similarity_search_with_score(query, k=k)

    # BM25 检索
    tokenized_docs = [doc.split() for doc in documents]
    bm25 = BM25Okapi(tokenized_docs)
    bm25_scores = bm25.get_scores(query.split())
    bm25_results = sorted(
        enumerate(bm25_scores),
        key=lambda x: x[1],
        reverse=True
    )[:k]

    # 融合（加权）
    fused = {}
    for doc, score in vec_results:
        fused[doc.page_content] = alpha * (1 - score)

    for idx, score in bm25_results:
        doc = documents[idx]
        if doc in fused:
            fused[doc] += (1 - alpha) * score
        else:
            fused[doc] = (1 - alpha) * score

    return sorted(fused.items(), key=lambda x: x[1], reverse=True)[:k]

import time
from collections import defaultdict

class RateLimiter:
    def __init__(self, max_requests=60, window=60):
        self.max = max_requests
        self.window = window
        self.requests = defaultdict(list)

    def allow(self, user_id):
        now = time.time()
        self.requests[user_id] = [
            t for t in self.requests[user_id]
            if now - t < self.window
        ]
        if len(self.requests[user_id]) >= self.max:
            return False
        self.requests[user_id].append(now)
        return True

class CostTracker:
    PRICES = {
        "deepseek-v4": {"input": 0.00028, "output": 0.0011},
        "deepseek-v4-pro": {"input": 0.0014, "output": 0.0056},
        "qwen-3.5": {"input": 0.0, "output": 0.0},
    }

    def __init__(self, budget=0.5):
        self.budget = budget
        self.spent = 0

    def record(self, model, tokens_in, tokens_out):
        price = self.PRICES[model]
        cost = (tokens_in * price["input"] +
                tokens_out * price["output"]) / 1000
        self.spent += cost
        return cost

    @property
    def remaining(self):
        return self.budget - self.spent

16. 手写：基于 LangGraph 的多 Agent 协作系统（工业级）⭐⭐⭐

要求：实现一个包含“研究员（Researcher）”和“写作者（Writer）”的流水线。要求用到状态机（State），并且能够演示如何在图节点之间传递信息。

核心考点：状态图（StateGraph）定义、节点路由逻辑、多实体状态流转。

可运行的完整解决方案与逐行讲解：

python 
from typing import TypedDict, Annotated, List
from langgraph.graph import StateGraph, END
from openai import OpenAI
import operator

# 初始化客户端
client = OpenAI(api_key="sk-xxx", base_url="https://api.deepseek.com/v1")

# 1. 定义全局状态字典（这是多Agent沟通的"黑板"）
# TypedDict 约束结构，Annotated 和 operator.add 解决列表追加而不是覆盖的问题
class AgentState(TypedDict):
    topic: str               # 用户指定的研究主题
    research_data: List[str] # 研究员收集的资料
    draft: str               # 写作者生成的初稿
    revision_count: int      # 审稿被打回的次数

# 2. 定义节点逻辑（每个节点就是一个函数，输入是State，输出也是State片段）
def researcher_node(state: AgentState):
    topic = state["topic"]
    print(f"🕵️  研究员正在搜索关于 [{topic}] 的资料...")
    # 模拟外部API调用或LLM推理
    prompt = f"请提供关于 {topic} 的3条核心事实，每条简短即可。"
    res = client.chat.completions.create(
        model="deepseek-v4",
        messages=[{"role": "user", "content": prompt}]
    )
    # 返回的内容会自动合并或覆盖到全局 state 中
    return {"research_data": [res.choices[0].message.content]}

def writer_node(state: AgentState):
    data = "\n".join(state["research_data"])
    print("✍️  写作者正在基于资料撰写草稿...")
    prompt = f"根据以下资料写一段100字左右的摘要：\n{data}"
    res = client.chat.completions.create(
        model="deepseek-v4",
        messages=[{"role": "user", "content": prompt}]
    )
    return {
        "draft": res.choices[0].message.content,
        # 记录写作次数
        "revision_count": state.get("revision_count", 0) + 1 
    }

def reviewer_node(state: AgentState):
    draft = state["draft"]
    print("🧐 审核员正在检查文章质量...")
    # 工业级系统中这里会让LLM进行判分，这里为了演示简化为字数判断
    if len(draft) < 50:
        print("❌ 审核不通过，打回重写。")
        return {"status": "rejected"}
    print("✅ 审核通过。")
    return {"status": "approved"}

# 3. 定义条件路由函数（决定下一步去哪里）
def review_router(state: AgentState) -> str:
    # 防止死循环的硬性保护机制
    if state.get("revision_count", 0) >= 3:
        return "end"
    
    # 根据审核员的隐式判断结果进行路由
    if len(state.get("draft", "")) < 50:
        return "rewrite"
    return "end"

# 4. 组装工作流图
workflow = StateGraph(AgentState)

# 添加节点
workflow.add_node("researcher", researcher_node)
workflow.add_node("writer", writer_node)
# 这里省略 reviewer_node 的直接调用，将逻辑融合到 router 中以简化图结构

# 定义边（控制流）
workflow.set_entry_point("researcher")
workflow.add_edge("researcher", "writer")

# 添加条件边（根据路由函数的结果决定下一步分支）
workflow.add_conditional_edges(
    "writer",
    review_router,
    {
        "rewrite": "writer", # 打回重写，回到 writer 节点
        "end": END           # 结束工作流
    }
)

# 5. 编译成可运行的应用
app = workflow.compile()

# 6. 执行示例
if __name__ == "__main__":
    initial_state = {
        "topic": "Agent技术的发展趋势",
        "research_data": [],
        "revision_count": 0
    }
    # stream 方法支持流式输出节点状态
    for s in app.stream(initial_state):
        print(s)
        print("---")

代码亮点总结：

  • 使用了 TypedDict 进行状态强类型校验，这是高标准工程的体现。
  • 包含了防死循环的兜底逻辑（revision_count >= 3），展示了安全编程意识。
  • 完整展示了 LangGraph 的三要素：State（状态）、Nodes（节点计算）、Edges（边与路由流转）。

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import os

app = FastAPI(title="Mini MCP Server")

# 1. 协议定义：MCP要求服务端必须返回支持的工具 Schema 列表
class ToolSchema(BaseModel):
    name: str
    description: str
    parameters: dict

class ToolListResponse(BaseModel):
    tools: list[ToolSchema]

# 2. 协议定义：执行工具的请求和响应结构
class ExecuteRequest(BaseModel):
    tool_name: str
    arguments: dict

class ExecuteResponse(BaseModel):
    result: str
    is_error: bool = False

# 3. 暴露工具发现接口 (Discovery)
@app.get("/tools", response_model=ToolListResponse)
def list_tools():
    """Agent 会首先调用这个接口，知道你有啥本事"""
    return ToolListResponse(tools=[
        ToolSchema(
            name="read_local_file",
            description="读取服务器上的文本文件内容",
            parameters={
                "type": "object",
                "properties": {
                    "file_path": {
                        "type": "string",
                        "description": "文件的绝对路径"
                    }
                },
                "required": ["file_path"]
            }
        )
    ])

# 4. 暴露工具执行接口 (Execution)
@app.post("/execute", response_model=ExecuteResponse)
def execute_tool(req: ExecuteRequest):
    """Agent 推理出需要调用的工具后，向这里发送指令"""
    if req.tool_name == "read_local_file":
        file_path = req.arguments.get("file_path")
        
        # 安全防御：防目录穿越攻击 (Directory Traversal)
        if ".." in file_path or not file_path.startswith("/tmp/safe_zone/"):
            return ExecuteResponse(
                result="Access Denied: 只能读取 /tmp/safe_zone/ 目录",
                is_error=True
            )
            
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                content = f.read()
            return ExecuteResponse(result=content[:2000]) # 截断防止Token溢出
        except Exception as e:
            return ExecuteResponse(result=str(e), is_error=True)
            
    raise HTTPException(status_code=404, detail="Tool not found")

18. 心智模型：工业级 Agent 的代码结构规范

在面试中，如果要求你白板设计项目结构，千万不要把所有东西塞进一个 main.py。优秀的 Agent 代码架构应该像微服务一样分层：

text 
├── agents/                 # Agent 核心定义层
│   ├── prompts/            # 模板文件单独管理 (jinja2/yaml)
│   ├── chains/             # LLM 调用链逻辑

2025-2026 最新进展

代码面试在 2026 年依然是开发岗的硬门槛，且"现场能跑"的要求愈发严格。高频手写题集中在三类：Agent 基础循环（LLM + 工具调用的 ReAct 实现）、RAG 链路（切分、嵌入、检索、重排）、以及工程实践（重试装饰器、并发调用、流式输出、token 计数与截断）。随着 Function Calling/Tool Use 成为标配，能否优雅地实现工具注册、参数校验与结果回填，成为区分水平的关键。
